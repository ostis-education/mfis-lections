\begin{frame}{\\Information Construct}
	\topline
	\justifying
	\begin{SCn}
		\scnheader{information construct}
		\scnidtf{construct (structure) that contains information about certain entities}
		\scnidtf{information model consisting of distinct \textit{signs} that denote \textit{entities} of any kind and, in particular, \textit{signs} that denote relationships between the signs of those \textit{entities}. Such relationships usually model relationships between the \textit{entities} denoted by the connected \textit{signs})}
		\scntext{note}{Information constructs may vary widely in how they are represented or materialized, how they are structured (their syntax), and what they \textit{mean} (their denotational semantics).}
	\end{SCn}

\end{frame}

\begin{frame}{\\Discrete Information Construct}
	\topline
	\justifying
	\begin{SCn}
		\footnotesize
		\scnheader{discrete information construct}
		\scnsubset{information construct}
		\scntext{explanation}{The meaning of each discrete information construct is determined by
		\begin{textitemize}
			 \item the set of its elements (syntactically atomic fragments)
			 \item the alphabet of those elements, a family of classes of syntactically equivalent elements of the information construct
			 \item the class of syntactically equivalent elements to which each element of the information construct belongs
			 \item the configuration of incidence relationships between the elements of the information construct
		 \end{textitemize}}
	\end{SCn}

\end{frame}

\begin{frame}{\\Non-Primary sc-Identifier}
	\topline
	\justifying
	Non-primary sc-identifiers list possible \textit{synonyms*} of the corresponding \textit{primary sc-identifier}. They can explain or even define the entity it denotes and indicate important properties of that entity.

	\bigskip
	For some sc-elements, both primary and non-primary sc-identifiers are used frequently, especially in informal texts such as explanations and notes. Identifying this class of sc-identifiers explicitly makes it easier to analyze the semantics of knowledge base source texts.
\end{frame}

\begin{frame}{\\Forming System sc-Identifiers}
	\topline
	\justifying
	\vspace*{\fill}\\
	\footnotesize
	A \textit{system sc-identifier} may contain Latin letters, digits, underscores, and hyphens. For internationalization, form \textit{system sc-identifiers} from the corresponding primary English \textit{sc-identifiers}.

	\bigskip
	To form a \textit{system sc-identifier} for an \textit{sc-element}, take its primary English identifier and replace characters outside that alphabet with underscores (\texttt{\_}). Uppercase letters are usually converted to lowercase.

	\bigskip
	For signs of \textit{role relations}, begin the \textit{system sc-identifier} with \texttt{rrel\_} in place of the \scnrolesign{} marker, then append the name of the \textit{role relation}.

\end{frame}

\begin{frame}{\\Forming System sc-Identifiers}
	\topline
	\justifying
	\vspace*{\fill}\\
	 For signs of \textit{non-role relations}, begin the \textit{system sc-identifier} with \texttt{nrel\_} in place of the asterisk, then append the name of the \textit{non-role relation}.\\

	\bigskip
	For signs of classes of \textit{concepts}, begin the \textit{system sc-identifier} with \texttt{concept\_}, then append the \textit{class name}.\\

	\bigskip
	For signs of \textit{structures}, begin the \textit{system sc-identifier} with \texttt{struct\_}, then append the \textit{structure name}.
\end{frame}

\begin{frame}{\\Non-Translatable sc-Identifier}
	\topline
	\justifying
	\vspace*{\fill}\\
	\footnotesize
	Non-translatable sc-identifiers are used only in \textit{knowledge base} source texts (including \textit{sc.s-texts}) and in messages exchanged between \textit{ostis systems}. They are used when several fragments of a source text or message must refer to the same \textit{sc-element} and it has no \textit{system sc-identifier} and does not need one.

	\bigskip
	\textit{Non-translatable sc-identifiers} make \textit{knowledge base} source texts easier to read and organize. They also let different source files or messages exchanged between \textit{ostis systems} refer to the same \textit{sc-element} that has no name within the knowledge base. Such \textit{sc-elements} often denote \textit{structures} or \textit{sc-connectors}.

	\bigskip
	Thus, \textit{non-translatable sc-identifiers} exist only outside an \textit{ostis system knowledge base}. No corresponding \textit{internal file} is created when the knowledge base is built from source texts or a received message is imported into it.
\end{frame}

\begin{frame}{\\Differences Between Identifier Types}
	\topline
	\justifying
	\vspace*{\fill}\\
	\small
	System identifiers must be unique across the entire knowledge base of the \textit{OSTIS Ecosystem}, regardless of the external language. They also use a simpler alphabet to facilitate automated processing.

	\bigskip
	\textit{System sc-identifiers} and \textit{non-translatable sc-identifiers} both serve to name \textit{sc-elements} in \textit{knowledge base} source texts or messages exchanged between \textit{ostis systems}.

	\bigskip
	Each \textit{system sc-identifier} is represented in the knowledge base as an \textit{internal file of the ostis system} and linked to its \textit{sc-element} by a pair of the \textit{system \mbox{sc-identifier*}} relation. \textit{Non-translatable sc-identifiers} have no representation, corresponding \textit{internal file}, or link to the identified \textit{sc-element} within the knowledge base.
\end{frame}

\begin{frame}{\\Simple sc-Identifier}
	\topline
	\justifying
	\vspace*{\fill}\\
	A simple sc-identifier identifies an sc-element. It contains no identifiers of other sc-elements and carries no information about the denoted entity that can be \textit{translated into SC-code}.

	\bigskip
	A simple string sc-identifier is a string of characters that names the same entity as the sc-element it identifies. Simple string sc-identifiers are the most common type assigned to sc-elements.
\end{frame}

\begin{frame}{\\Simple sc-Identifier}
	\topline
	\justifying
	\vspace*{\fill}\\
	\small{
		\textit{sc-identifiers} for external files of ostis systems specify the location of those files. Each is a string formed according to the URL standard and enclosed in double quotes. The quotes delimit the sc-identifier unambiguously because a URL can contain spaces in this notation. This is useful because such sc-identifiers often occur in ostis system knowledge base source files.

		\bigskip
		\textit{sc-identifiers} for external files of ostis systems are simple string sc-identifiers in OSTIS Technology, even though they may contain special characters such as "\%"{} or "/"{}. They identify the element's location in the current state of the system's environment without conveying semantic information about the element's properties.
	}
\end{frame}

\begin{frame}{\\Complex sc-Identifier}
	\topline
	\justifying
	\vspace*{\fill}\\
	\scriptsize
		\textbf{\textit{sc-expressions}} fall into the following categories:
		\begin{textitemize}
			\item \textit{sc-expression for an unordered set} -- an sc-expression enclosed in curly braces
			\item \textit{sc-expression for a structure} -- an sc-expression denoting a structure represented in any known, readily identifiable language (Russian, English, SCg-code, SCs-code, or SCn-code)

			\textit{sc-expression for a structure} denotes a structure containing an sc-text semantically equivalent to the text in curly braces, which is written in a known language. Usually the text is written in a formal language such as SCs-code and can be interpreted automatically and unambiguously. A text in an informal language such as Russian is much harder to interpret automatically and may be ambiguous.

			The current knowledge base authoring tools follow an older version of the sc-expression rules, in which an \textit{sc-expression for a structure} is delimited by square brackets with asterisks ("[*"{} and "*]"{}).

			\item \textit{sc-expression for an ordered set} -- an sc-expression for a tuple enclosed in angle brackets. It denotes an ordered set of sc-elements whose order is given by the sequence of their sc-identifiers.
		\end{textitemize}
\end{frame}

\begin{frame}{\\Complex sc-Identifier}
	\topline
	\justifying
	\vspace*{\fill}\\
	\scriptsize
		\begin{textitemize}
			\item \textit{sc-expression for an internal file of an ostis system} -- an sc-expression denoting an \textit{internal file of an ostis system} whose representation is enclosed in square brackets. The contents of the \textit{internal file} appear between those brackets.

			Further details about an \textit{internal file of an ostis system} can readily be specified in \textit{SC-code}, such as the \textit{language} of the \textit{information construct} that forms the \textit{file's} contents, the encoding format, the \textit{author}*, and more.

			\item \textit{sc-expression for a sample file of an ostis system} -- an sc-expression delimited by square brackets with exclamation marks.

			\item \textit{sc-expression based on a binary relation} -- an sc-expression consisting of either (1) the \textit{sc-identifier} of a binary directed relation followed by (2) the sc-identifier of an element of its first domain in parentheses, or (1) the sc-identifier of a binary undirected relation followed by (2) the sc-identifier of an element of its domain in parentheses. The expression thus specifies a binary relation, typically a function, and one of its arguments (in parentheses).
		\end{textitemize}
\end{frame}

\begin{frame}{\\Complex sc-Identifier}
	\topline
	\justifying
	\vspace*{\fill}\\
	\scriptsize{
		\begin{textitemize}
			\item{\textit{sc-expression based on an algebraic operation} -- a parenthesized sc-expression in which the algebraic operator separates the \textit{sc-identifiers}.}
			\item{\textit{sc-expression identifying an sc-connector} -- a parenthesized expression for the \textit{sc-connector} incident to the two given sc-elements. The corresponding sc.s-connector symbol specifies its type.
				To make \textit{sc-expressions identifying sc-connectors} easier to read and process, their first and third components must be \textit{simple sc-identifiers}. In sc.s-texts, these expressions may also contain sc.s-modifiers.}
		\end{textitemize}
