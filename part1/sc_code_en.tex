\begin{frame}{\\Von Neumann Architecture}
	\topline
	\justifying
	\large
	\begin{textitemize}
		\item computer memory is homogeneous (linear)
		\item computer memory locations are accessed by address
		\item the alphabet for representing data and instructions is the set {0, 1}
		\item each program consists of a sequence of instructions executed by the processor
		\item programs can include conditional branch instructions
	\end{textitemize}
\end{frame}

\begin{frame}{\\Proposed Approach}
	\topline
	\justifying
	\large
	Develop a method of encoding information that

	\begin{textitemize}
		\item is easy for a computer to store and process
		\item is easy for people to understand
		\item conveys meaning and can be interpreted unambiguously
	\end{textitemize}

	\bigskip
	We need to move from \textit{data} to \textit{knowledge}.

	\bigskip
	A key distinguishing feature of knowledge\\ is its \uline{interpretability} (D. A. Pospelov).

\end{frame}

\begin{frame}{\\Principles of SC-code}
	\topline
	\justifying
	\vspace{8mm}
	\begin{SCn}
		\scnheader{SC-code}
		\scntext{explanation}{Information is carried by the configuration of connections between the signs}
		\scniselement{abstract language}
		\scniselement{graph-based language}
		\begin{scnrelfromset}{basic principles}
			\scnfileitem{signs of entities are syntactically elementary fragments of sc-texts and have no internal structure}
			\scnfileitem{information constructs other than sc-texts (texts in SC-code) can be included in the knowledge base as files}
			\scnfileitem{texts in SC-code are nonlinear because each sign of an entity appears only once in the knowledge base and can have any number of connections to other signs}
		\end{scnrelfromset}
	\end{SCn}
\end{frame}

\begin{frame}{\\}
	\topline
	\justifying
	\begin{SCn}
		\scnheader{SC-code}
		\begin{scnrelfromset}{basic principles}
			\scnfileitem{the knowledge base is a graph structure whose alphabet includes nodes, edges, arcs, basic arcs, and special nodes representing files)}
			\scnfileitem{arcs and edges in a knowledge base graph can connect nodes to other nodes, edges, or arcs, and so on.}
			\scnfileitem{identical sc-elements are merged to eliminate synonymy}
			\scnfileitem{sc-nodes, sc-edges, and sc-arcs denote different entities}
		\end{scnrelfromset}
	\end{SCn}
\end{frame}

\begin{frame}{\\Syntax and the SC-code Alphabet}
	\topline
	\justifying
	\begin{SCn}
		\scnheader{SC-code alphabet}
		\scnidtf{family of syntactic tags assigned to sc-elements to indicate the corresponding classes to which they belong}

		\scnheader{minimal SC-code alphabet}
		\scntext{explanation}{To read sc-constructs in sc-memory, it suffices to identify \textit{basic sc-arcs} by their syntax. They are constant, permanent, positive sc-membership pairs. They connect each sc-element to its sc-classes. Basic sc-arcs themselves cannot be identified this way.}
	\end{SCn}
\end{frame}

\begin{frame}{\\Minimal SC-code Alphabet}
	\topline
	\justifying
	\begin{SCn}
		Thus any class of sc-elements can be identified explicitly by:\\
		\begin{textitemize}
			\item{introducing an sc-element that serves as the sign of that class of sc-elements (an sc-class);}
			\item{drawing permanent positive sc-membership pairs to all sc-elements that belong to the
			sc-class in question and are present in the current state of sc-memory.}
		\end{textitemize}

		\scnheader{minimal SC-code alphabet}
		\scnidtf{\textit{Class of basic sc-arcs} and \textit{class of all other sc-elements} (by default)}
	\end{SCn}
\end{frame}

\begin{frame}{\\Semantic Classification of sc-Elements}
	\topline
	\justifying
	\textbf{}\\
	The main criteria for classifying \textit{sc-elements} include:
	\begin{textitemize}
		\item structure;
		\item logical semantics;
		\item temporal properties of the entities denoted by sc-elements, including:
		\begin{textitemize}
			\item whether the denoted entity exists permanently or temporarily;
			\item whether the denoted entity is static (stationary) or dynamic (changeable.
		\end{textitemize}
	\end{textitemize}
\end{frame}

\begin{frame}{\\Temporal Properties of sc-Elements}%Possibly redundant
	\topline
	\justifying
	\vspace{10mm}
	\begin{SCn}
		\textbf{When discussing the temporal properties of sc-elements, distinguish between:}
		\begin{textitemize}
			\item the temporary presence of an individual sc-element in the knowledge base that contains it
			\item the temporary presence of an entire sc-construct in sc-memory (a situation in sc-memory);
			\item the temporary existence of the external entity denoted by an sc-element;
			\item whether the external entity denoted by an sc-element is static or dynamic. A dynamic entity entails a description in sc-memory of the process by which its state or configuration changes;
			\item dynamic sc-set representing the corresponding external process;
			\item dynamic sc-set representing the corresponding internal process
		\end{textitemize}
	\end{SCn}
\end{frame}
