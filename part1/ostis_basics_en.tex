\begin{frame}{\\Cybernetic System}
	\topline
	\justifying

\vspace{10pt}
\begin{SCn}
	\footnotesize
	\scnheader{cybernetic system}
	\scnidtf{material entity capable of acting purposefully on its environment at least to preserve its integrity, viability, and safety}
	\scnidtf{system whose operation is organized around processing information about its environment}
	\scnidtf{material entity capable of active, goal-directed behavior that, at a certain level of development, becomes conscious, planned, and intentional}
	\scnidtf{agent capable of independently performing certain internal and external actions assigned to it or initiated on its own}
	\scnidtf{natural or artificial system capable of monitoring and analyzing its own state and its environment and actively influencing both}
	\scnidtf{system capable of interacting with its environment with sufficient autonomy to solve a variety of problems}
	\scnidtf{system based on information processing}
\end{SCn}
\end{frame}

\begin{frame}{\\Complex Problem}
	\topline
	\justifying
	\begin{SCn}
		\scnheader{complex problem}
		\scnidtf{problem that requires different kinds of knowledge and different problem-solving models}
		\scntext{note}{The set of problem-solving models needed for a complex problem cannot be determined in advance}
		\vspace{5mm}
		\textbf{Examples of Complex Problems:}
		\begin{textitemize}
			\item understanding natural languages, images, and spoken messages
			\item planning the behavior of intelligent robots
			\item comprehensive enterprise automation
			\item adaptive instruction in different subjects
		\end{textitemize}
	\end{SCn}
\end{frame}

\begin{frame}{\\Hybrid Intelligent System}
	\topline
	\justifying
	\vspace{10mm}
	\begin{SCn}
		\scnheader{hybrid intelligent system}
		\scnidtf{intelligent system combining multiple kinds of knowledge with multiple problem-solving models}
		\scntext{note}{Hybrid intelligent systems are designed to solve complex problems}
		\begin{scnrelfromset}{shortcomings of current systems}
				\scnfileitem{monolithic design}
				\scnfileitem{focus on a single class of problems (rather than multiple classes)}
				\scnfileitem{enormous resources, including time, needed for development}
				\scnfileitem{monolithic design prevents reuse of components for other problems}
		\end{scnrelfromset}
	\end{SCn}
\end{frame}

\begin{frame}{\\Technology OSTIS}
	\topline
	\justifying
	\vspace{10mm}
	\begin{SCn}
		\scnheader{OSTIS}
		\scnidtf{Open Semantic Technology for Intelligent Systems}
		\scnidtf{open, integrated technology for designing interoperable intelligent systems}
		\begin{scnrelfromset}{basic principles}
			\scnfileitem{an OSTIS knowledge base can describe \underline{any kind} of knowledge}
			\scnfileitem{an OSTIS problem solver uses a multi-agent approach and makes it easy to \underline{combine any} problem-solving models}
			\scnfileitem{the interface of an ostis system is a subsystem with its own knowledge base and problem solver (it can also be described in SC-code)}
			\scnfileitem{the technology uses a \underline{universal} way of representing (encoding) information called SC-code}
		\end{scnrelfromset}
	\end{SCn}
\end{frame}

\begin{frame}{\\}
	\topline
	\justifying
	\vspace{10mm}
	\begin{SCn}
	\scnheader{OSTIS}
	\begin{scnrelfromset}{advantages}
		\scnfileitem{uniform representation (\underline{the same form} for all information)}
		\scnfileitem{easy for machines to process and for people to understand}
		\scnfileitem{any knowledge and problem-solving models can be integrated into an ostis system on a plug-and-play basis, and the system can always be retrained}
		\scnfileitem{universal, interoperable components (reuse can reduce development time for new components by 40--60\%)}
		\scnfileitem{the system is described in SC-code, so it can \underline{analyze itself}, find its own errors, and optimize its operation (reflection)}
		\end{scnrelfromset}
	\end{SCn}
\end{frame}

\begin{frame}{\\}
	\topline
	\justifying
	\vspace{10mm}
	\begin{SCn}
		\scnheader{OSTIS}
		\begin{scnrelfromset}{advantages}
			\scnfileitem{platform independence (development is independent of computer architecture, and the platform can be implemented in software or hardware)}
			\scnfileitem{parallel information processing}
			\scnfileitem{an ostis system can incorporate components developed with OSTIS, connect to other systems, and integrate other components (using JSON or an API)}
			\scnfileitem{an ostis system can match the performance of a conventional system and sometimes outperform it through parallel processing (semantic computers would improve performance further)}
		\end{scnrelfromset}
	\end{SCn}
\end{frame}

\begin{frame}{\\}
	\topline
	\justifying
	\vspace{10mm}
	\begin{SCn}
		\textbf{What OSTIS Provides:}
			\begin{textitemize}
			\item  {OSTIS -- a \textbf{technology for developing} intelligent systems, each of which addresses a particular class of problems}
			\item {The principal advantage of OSTIS is how readily developers can \textbf{modify and extend} systems, adapt them to new problems, and build up a collection of reusable components that reduce the time and effort needed to develop new systems. Most ostis system functions can also be implemented with conventional tools}
			\item {OSTIS -- a way to address \textbf{interoperability}, one of the major challenges in modern technology}
			\end{textitemize}
	\end{SCn}
\end{frame}

\begin{frame}{\\Information Construct}
	\topline
	\justifying
	\begin{SCn}
		\scnheader{information construct}
		\scnidtf{construct (structure) that contains information about certain entities}
		\scnidtf{information}
		\scntext{note}{An information construct has
			\begin{textitemize}
				\item a form of representation (text, sound, image)
				\item a structural form (syntax)
				\item a meaning (denotational semantics)
			\end{textitemize}
		}
	\end{SCn}
\end{frame}

\begin{frame}{\\}
	\topline
	\justifying
		\begin{SCn}
		\scnheader{information construct}
		\begin{scnrelfromset}{partition}
			\scnitem{sc-set}
				\begin{scnindent}
				\scnidtf{internal information construct of an ostis system stored in its internal sc-memory}
				\end{scnindent}
			\scnitem{file}
				\begin{scnindent}
				\scnidtf{internal information construct of an ostis system stored in its external file storage}
				\scntext{note}{A file may be stored in another system's memory}
				\end{scnindent}
			\scnitem{external information construct that is neither a file nor an sc-construct}
		\end{scnrelfromset}
	\scnsuperset{discrete information construct}
	\end{SCn}
\end{frame}

\begin{frame}{\\Discrete Information Construct}
	\topline
	\justifying
	\begin{SCn}
		\small
		\scnheader{discrete information construct}
\scntext{explanation}{The meaning of each discrete information construct is determined by:
	\begin{textitemize}
		\item the set of its elements (syntactically atomic fragments)
		\item the alphabet of those elements, a family of classes of syntactically equivalent elements of the information construct
		\item the class of syntactically equivalent elements to which each element of the information construct belongs
		\item the configuration of incidence relationships between the elements of the information construct.
	\end{textitemize}	}
	\end{SCn}
\end{frame}

\begin{frame}{\\}
	\topline
	\justifying
	\begin{SCn}
		\small
		\scnheader{discrete information construct}
		\scntext{note}{To analyze the meaning of a discrete information construct, we need not specify how its elements are represented. What matters is:
		\begin{textitemize}
			\item a simple procedure for identifying (segmenting) the elements of a discrete information construct
			\item a simple procedure for determining whether different elements of a discrete information construct are syntactically equivalent
			\item a simple procedure for assigning each element of a discrete information construct to its class of syntactically equivalent elements (its corresponding alphabet element).
		\end{textitemize}
			}
	\end{SCn}
\end{frame}

\begin{frame}{\\Formal Language}
	\topline
	\justifying
	\begin{SCn}
		\scnheader{language}
		\scnidtf{set of information constructs built according to common syntactic and semantic rules}
		\begin{scnrelfromset}{partition}
			\scnitem{artificial (constructed) language}
			\scnitem{natural language}
		\end{scnrelfromset}
		\scnsuperset{formal language}

		\scnheader{formal language}
		\scnidtf{language in which the meaning of each word or sign and the rules for forming and interpreting sentences are unambiguous}
		\scnidtf{set of constructs formed from a finite alphabet of the language}
	\end{SCn}

\begin{frame}{\\Sign}
	\topline
	\justifying
	\begin{SCn}
		\scnheader{sign}
		\scnidtf{fragment of an information construct that \underline{denotes} an entity (object) described by that information construct alongside other entities}

		\begin{scnrelfromset}{partition}
			\scnitem{sign that is an element of a \textit{discrete information construct}}
			\scnitem{sign that is a non-atomic fragment of a \textit{discrete information construct}}
		\end{scnrelfromset}

	\end{SCn}
\end{frame}

\begin{frame}{\\}
	\topline
	\justifying

A \textbf{\textit{sign}} is a fragment of an \textit{information construct} that conventionally represents (depicts) a described entity called the \textbf{\textit{denotatum of the sign}}.

\bigskip

The absence of a \textit{sign} denoting an entity means that we are unaware of the entity and have yet to study it. The entity may still exist.

\end{frame}

	\scnitem{semantics (studies the meanings of signs)}

	\end{SCn}

	\justifying

	\scnheader{semantics}
	\scnidtf{study of relationships between signs and their meanings}
	\scntext{note}{The word semantics refers to signification or meaning}
